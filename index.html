        function openPresentation(category) {

            currentPresSlides = evalData.filter(item => item.category === category).sort((a,b) => {

                const matchA = a.title.match(/^(\d+(\.\d+)*)/); const matchB = b.title.match(/^(\d+(\.\d+)*)/);

                const numA = matchA ? matchA[1].split('.').map(Number) : []; const numB = matchB ? matchB[1].split('.').map(Number) : [];

                for (let i = 0; i < Math.max(numA.length, numB.length); i++) {

                    if ((numA[i] || 0) < (numB[i] || 0)) return -1;

                    if ((numA[i] || 0) > (numB[i] || 0)) return 1;

                } return 0;

            });

'active border-emerald-400 bg-slate-800 text-white shadow-md' : 'border-transparent text-slate-400 hover:bg-slate-800/50 hover:text-slate-200'}">

                    <div class="line-clamp-2 font-bold">${slide.title}</div>
